function subscribeRealtime() {

  const blockCh = sb.channel('blocks-' + SLUG)

    .on('postgres_changes', { event:'*', schema:'public', table:'blocks', filter:`event_slug=eq.${SLUG}` },

      handleBlockChange)

    .subscribe();


  const commentCh = sb.channel('comments-' + SLUG)

    .on('postgres_changes', { event:'INSERT', schema:'public', table:'comments' },

      p => handleNewComment(p.new))

    .subscribe();


  const presenceCh = sb.channel('presence-' + SLUG)

    .on('postgres_changes', { event:'*', schema:'public', table:'presence', filter:`slug=eq.${SLUG}` },

      refreshOnline)

    .subscribe();


  channels = [blockCh, commentCh, presenceCh];

}


function handleBlockChange({ eventType, new:row, old }) {

  for (const tab of ['evjf','mariage','jourj']) {

    if (eventType === 'DELETE') {

      blocks[tab] = blocks[tab].filter(b => b.id !== old?.id);

    } else if (row?.event_slug === SLUG) {

      const existing = blocks[tab].find(x => x.id === row.id);

      const updated  = { id:row.id, type:row.type, role:row.role, title:row.title,

                         comments: existing?.comments || [], ...row.content };

      if (row.tab === tab) {

        if (existing) Object.assign(existing, updated);

        else blocks[tab].push(updated);

      } else if (existing) {

        blocks[tab] = blocks[tab].filter(x => x.id !== row.id);

      }

    }

  }

  renderAll();

}


function handleNewComment(row) {

        el.innerHTML = '💬 Commentaires <span class="ctcount">' + b.comments.length + '</span>';

  }

}

function smartRender(text) {

  const frag = document.createDocumentFragment();

  text.split('\n').forEach((line, li, arr) => {

    frag.appendChild(smartRenderLine(line));

    if (li < arr.length - 1) frag.appendChild(document.createElement('br'));

  });

  return frag;

}


function smartRenderLine(line) {

  const span = document.createElement('span');

  const stripped = line.replace(/^[\p{Emoji}\s]+/u, '').trim();

  const phoneRx = /^(?:\+33|0)[0-9](?:[\s.\-]?[0-9]{2}){4}$/;

  const emailRx = /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/;

  const webRx   = /^https?:\/\//i;

  const addrRx  = /(rue|avenue|boulevard|place|chemin|allée|domaine|château|route|hôtel|hotel|lieu\s*:)/i;


  if (webRx.test(stripped)) {

    const prefix = line.replace(stripped, '');

    if (prefix) span.appendChild(document.createTextNode(prefix));

    span.appendChild(makeLink('smart-link web', stripped, '🔗 ' + stripped.replace(/^https?:\/\//, ''), stripped, true));

  } else if (emailRx.test(stripped)) {

    span.appendChild(makeLink('smart-link email', 'mailto:' + stripped, '✉️ ' + stripped));

  } else if (phoneRx.test(stripped)) {

    span.appendChild(makeLink('smart-link', 'tel:' + stripped.replace(/[\s.\-]/g, ''), '📞 ' + stripped));

  } else if (addrRx.test(stripped)) {

    const colonIdx = stripped.indexOf(':');

    const addr = colonIdx >= 0 ? stripped.slice(colonIdx + 1).trim() : stripped;

    const prefix = colonIdx >= 0 ? line.slice(0, line.indexOf(':') + 1) + ' ' : '';

    if (prefix) span.appendChild(document.createTextNode(prefix));

    span.appendChild(makeLink('smart-link addr', 'https://maps.google.com/?q=' + encodeURIComponent(addr), '📍 ' + addr, null, true));

  } else {

    span.textContent = line;

  }

  return span;

}


function renderCellContent(cell) {

  if (!cell) return document.createTextNode('');

  const stripped = cell.trim();

  const phoneRx = /^(?:\+33|0)[0-9](?:[\s.\-]?[0-9]{2}){4}$/;

  const emailRx = /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/;

  const webRx   = /^https?:\/\//i;

  const addrRx  = /(rue|avenue|boulevard|place|chemin|domaine|château|route)/i;

  if (phoneRx.test(stripped)) return makeLink('smart-link', 'tel:' + stripped.replace(/[\s.\-]/g,''), '📞 ' + stripped);

  if (emailRx.test(stripped)) return makeLink('smart-link email', 'mailto:' + stripped, '✉️ ' + stripped);

  if (webRx.test(stripped))   return makeLink('smart-link web', stripped, '🔗 ' + stripped.replace(/^https?:\/\//, ''), null, true);

  if (addrRx.test(stripped))  return makeLink('smart-link addr', 'https://maps.google.com/?q=' + encodeURIComponent(stripped), '📍 ' + stripped, null, true);

  const span = document.createElement('span'); span.textContent = cell; return span;

}


function makeLink(cls, href, text, title, newTab) {

  const a = document.createElement('a');

  a.className = cls; a.href = href; a.textContent = text;

  if (title) a.title = title;

  if (newTab) { a.target = '_blank'; a.rel = 'noopener'; }

  a.addEventListener('click', e => e.stopPropagation());

  return a;

}
